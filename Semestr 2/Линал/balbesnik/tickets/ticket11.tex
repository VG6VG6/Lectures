\section{Определение линейной оболочки системы векторов}

\begin{definition}[Линейная оболочка]
\textbf{Линейной оболочкой} системы векторов $\alpha_1, \ldots, \alpha_k \in V$ называется множество
\[
L(\alpha_1, \ldots, \alpha_k) = \{ x \in V : x = \lambda_1 \alpha_1 + \cdots + \lambda_k \alpha_k,\ \lambda_i \in F \}.
\]
\end{definition}
