\section{Определение собственного вектора и собственного числа}

\begin{definition}[Собственный вектор и собственное число]
Пусть $V, F$ --- линейное пространство над полем $F$, $\varphi \in L(V, V)$, $E$ --- базис в $V$.

Вектор $x \neq 0$ называется \textbf{собственным вектором} оператора $\varphi$, соответствующим \textbf{собственному числу} $\lambda$, если
\[
\varphi(x) = \lambda x,
\]
или эквивалентно
\[
(\varphi - \lambda I)(x) = 0.
\]

\begin{remark}
$\lambda$ может равняться $0$, в этом случае $x \in \operatorname{Ker} \varphi$.
\end{remark}
\end{definition}
