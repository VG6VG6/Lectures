\section{Определение подпространства}

\begin{definition}[Подпространство]
Пусть $V$ --- линейное пространство над полем $F$, $L \subseteq V$ --- подмножество. $L$ называется \textbf{подпространством} $V$, если:
\begin{enumerate}
    \item $L \neq \emptyset$;
    \item $\forall a, b \in L,\ \forall \lambda, \mu \in F: \quad \lambda a + \mu b \in L$.
\end{enumerate}
Если $L \subset V$ ($L$ не совпадает с $V$), то $L$ называется \textbf{собственным подпространством}.
\end{definition}
