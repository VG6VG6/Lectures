\section{Определение базиса в линейном пространстве}

\begin{definition}[Базис линейного пространства]
Упорядоченный набор векторов $E = \{e_1, \ldots, e_k\}$ из $V$ называется \textbf{базисом} в $V$, если:
\begin{enumerate}
    \item Векторы $e_1, \ldots, e_k$ линейно независимы;
    \item Для любого $x \in V$ существуют числа $\xi_1, \ldots, \xi_k \in F$ такие, что $x = \xi_1 e_1 + \cdots + \xi_k e_k$.
\end{enumerate}
\end{definition}
