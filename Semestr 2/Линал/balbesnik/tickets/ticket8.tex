\section{Формулировка основной леммы о линейной зависимости}

\begin{lemma}[Основная лемма о линейной зависимости]
Пусть $S_1 = \{a_1, \ldots, a_r\} \subset V$ и $S_2 = \{b_1, \ldots, b_k\} \subset V$. Если:
\begin{enumerate}
    \item $S_1$ --- линейно независимая система;
    \item каждый вектор из $S_1$ линейно выражается через векторы из $S_2$, т.е. $a_i = d_{1i} b_1 + \cdots + d_{ki} b_k$,
\end{enumerate}
то $r \leq k$.
\end{lemma}
