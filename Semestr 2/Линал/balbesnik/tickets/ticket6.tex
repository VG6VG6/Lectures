\section{Определение размерности линейного пространства}

\begin{definition}[Размерность линейного пространства]
\textbf{Размерность} $V$ равна $n$ ($\dim V = n$), если:
\begin{enumerate}
    \item В $V$ существуют $n$ линейно независимых векторов;
    \item Любой набор из $n+1$ векторов в $V$ линейно зависим.
\end{enumerate}
\end{definition}
