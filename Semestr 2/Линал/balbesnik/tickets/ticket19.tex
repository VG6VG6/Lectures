\section{Определение ядра и образа линейного оператора}

\begin{definition}[Ядро и образ линейного оператора]
Пусть $\varphi \in L(V, V)$.

\textbf{Ядро} оператора:
\[
\operatorname{Ker} \varphi = \{ x \in V : \varphi(x) = 0 \}.
\]

\textbf{Образ} оператора:
\[
\operatorname{Im} \varphi = \{ y \in V : \exists x \in V,\ y = \varphi(x) \}.
\]
\end{definition}
