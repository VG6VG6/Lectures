\section{Определение линейной зависимости и линейной независимости векторов}

\begin{definition}[Линейная зависимость]
Векторы $\alpha_1, \ldots, \alpha_k$ называются \textbf{линейно зависимыми} (ЛЗ), если существуют числа $\lambda_1, \ldots, \lambda_k$, не все равные нулю, такие что
\[
\lambda_1 \alpha_1 + \cdots + \lambda_k \alpha_k = 0.
\]
\end{definition}

\begin{definition}[Линейная независимость]
Векторы $\alpha_1, \ldots, \alpha_k$ называются \textbf{линейно независимыми} (ЛНЗ), если из равенства
\[
\lambda_1 \alpha_1 + \cdots + \lambda_k \alpha_k = 0
\]
следует, что все $\lambda_i = 0$.
\end{definition}
